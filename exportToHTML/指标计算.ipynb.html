109  class DMysqlPoolConn:
110  
111      __pool = None
112  
113      def __init__(self, config):
114  
115          if not self.__pool:
116              self.__class__.__pool = PooledDB(creator=pymysql,
117                                               maxconnections=config.maxConnection,
118                                               mincached=config.minCached,
119                                               maxcached=config.maxCached,
120                                               maxshared=config.maxShared,
121                                               blocking=config.blocking,
122                                               maxusage=config.maxUsage,
123                                               setsession=config.setSession,
124                                               #charset=config.charset,
125                                               host=config.host,
126                                               port=config.port,
127                                               database=config.db,
128                                               user=config.user,
129                                               password=config.password,
130                                               )
131  
132      def get_conn(self):
133          return self.__pool.connection()

955  def get_ROR(row):
956      import math
957      a = row['DE']
958      b = row['De']
959      c = row['dE']
960      d = row['de']    
961      ror = a/c/(b/d)
962      ci_lower = math.log(ror) - 1.96 * math.sqrt(1/a + 1/b + 1/c + 1/d)
963      ci_upper = math.log(ror) + 1.96 * math.sqrt(1/a + 1/b + 1/c + 1/d)
964      return a/c/(b/d), math.exp(ci_lower), math.exp(ci_upper)

967  
968      a = row['DE']
969      b = row['De']
970      c = row['dE']
971      d = row['de']    
972      prr = a/(a+b)/(c/(c+d))
973      ci_lower = math.log(prr) - 1.96 * math.sqrt(1/a - 1/(a+b) + 1/c - 1/(c+d))
974      ci_upper = math.log(prr) + 1.96 * math.sqrt(1/a - 1/(a+b) + 1/c - 1/(c+d))
975      return prr, math.exp(ci_lower), math.exp(ci_upper)
